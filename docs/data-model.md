# Data Model (core tables)

- organizations(id, name, settings)
- users(id, org_id, email, role, mfa)
- assets(id, org_id, name, type, host, port, tags, last_seen)
- credentials(id, org_id, name, type, vault_ref)
- benchmarks(id, slug, title, technology, version, profile_levels, source_pdf_path)
- sections(id, benchmark_id, parent_id, number, title)
- controls(id, section_id, cis_id, title, description, rationale, remediation, scoring, profile)
- checks(id, control_id, type, command, expected, regex, severity)
- scan_jobs(id, org_id, benchmark_id, profile, status, created_by, created_at)
- scan_tasks(id, job_id, asset_id, status, started_at, finished_at)
- results(id, task_id, check_id, status, actual, expected, evidence_ref, score_impact)
- exceptions(id, control_id, asset_id, reason, expires_at, approved_by)
- reports(id, job_id, format, storage_ref, created_at)
